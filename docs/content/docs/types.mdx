## MiteConfig

Configuration for the Mite instance.

```ts
interface MiteConfig {
  apiKey?: string
  endpoint?: string
  timeout?: number
  retries?: number
  anonymousId?: string
  identityStorage?: MiteIdentityStorage
  identificationOptOut?: boolean
}
```

## MiteIdentityStorage

Storage adapter interface (compatible with `AsyncStorage`):

```ts
interface MiteIdentityStorage {
  getItem(key: string): Promise<string | null>
  setItem(key: string, value: string): Promise<void>
  removeItem(key: string): Promise<void>
}
```

interface Release {

## FeatureRequest

```ts
interface FeatureRequest {
  id: string
  title: string
  description?: string
  status: FeatureRequestStatus
  author_name?: string
  author_email?: string
  vote_count: number
  createdAt: number
}

type FeatureRequestStatus = 'OPEN' | 'IN_PROGRESS' | 'COMPLETED' | 'CLOSED'
```
